// SplitDuo Database Schema
// Use this DBML at dbdiagram.io to visualize the database structure

Project SplitDuo {
  database_type: 'PostgreSQL'
  Note: '''
    Database schema for SplitDuo - Expense Splitting App
    - Code-first approach with Entity Framework
    - All entities are auditable (created_at, updated_at)
    - Integer IDs for internal use, GUIDs for frontend
  '''
}

Table users {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  email varchar(255) [unique, not null]
  password_hash varchar(255) [not null, note: 'Hashed password']
  first_name varchar(100) [not null]
  last_name varchar(100)
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  Note: 'User accounts - minimal custom implementation without Identity'
}

Table groups {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  name varchar(200) [not null]
  description text
  created_by_user_id integer [ref: > users.id, not null]
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  Note: 'Groups for expense sharing - initially focused on couples (2 users)'
}

Table group_members {
  id integer [primary key, increment]
  group_id integer [ref: > groups.id, not null]
  user_id integer [ref: > users.id, not null]
  role_id int [not null, default: 'member', note: 'member = 2, admin = 1']
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  indexes {
    (group_id, user_id) [unique]
  }

  Note: 'Group membership - links users to expense groups'
}

Table expenses {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  group_id integer [ref: > groups.id, not null]
  title varchar(255) [not null]
  description text
  amount decimal(12,2) [not null, note: 'Total expense amount']
  paid_by_user_id integer [ref: > users.id, not null, note: 'User who paid the expense']
  expense_date date [not null, note: 'Date only (maps to .NET DateOnly)']
  category_id int [not null, note: 'Optional categorization']
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  Note: 'Individual expenses within a group'
}

Table expense_splits {
  id integer [primary key, increment]
  expense_id integer [ref: > expenses.id, not null]
  user_id integer [ref: > users.id, not null]
  split_amount decimal(12,2) [not null, note: 'Amount this user owes for this expense']
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]

  indexes {
    (expense_id, user_id) [unique]
  }

  Note: 'How each expense is split between users'
}

Table settlements {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  group_id integer [ref: > groups.id, not null]
  from_user_id integer [ref: > users.id, not null, note: 'User making the payment']
  to_user_id integer [ref: > users.id, not null, note: 'User receiving the payment']
  amount decimal(12,2) [not null, note: 'Settlement amount']
  settlement_date date [not null, note: 'Date only (maps to .NET DateOnly)']
  description text [note: 'Optional settlement description']
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  Note: 'Settlements between users to balance expenses'
}

Table imports {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  user_id integer [ref: > users.id, not null, note: 'User who performed the import']
  group_id integer [ref: > groups.id, not null, note: 'Group where data was imported']
  filename varchar(255) [not null, note: 'Original backup filename']
  import_date date [not null, note: 'Date only (maps to .NET DateOnly)']
  records_imported integer [not null, default: 0]
  status varchar(50) [not null, default: 'pending', note: 'pending, completed, failed']
  error_details text [note: 'Error information if import failed']
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]

  Note: 'Track import operations for audit purposes'
}
