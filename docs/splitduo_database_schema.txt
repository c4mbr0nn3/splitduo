// SplitDuo Database Schema
// Use this DBML at dbdiagram.io to visualize the database structure

Project SplitDuo {
  database_type: 'PostgreSQL'
  Note: '''
    Database schema for SplitDuo - Expense Splitting App
    - Code-first approach with Entity Framework
    - All entities are auditable (created_at, updated_at)
    - Integer IDs for internal use, GUIDs for frontend
  '''
}

Table users {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  email varchar(255) [unique, not null]
  password_hash varchar(255) [not null, note: 'Hashed password']
  first_name varchar(100) [not null]
  last_name varchar(100)
  two_factor_enabled boolean [default: false, not null, note: '2FA activation status']
  totp_secret varchar(255) [null, note: 'Base32 encoded TOTP secret for authenticator apps']
  backup_codes text [null, note: 'JSON array of hashed emergency backup codes']
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  Note: 'User accounts with 2FA support - minimal custom implementation without Identity'
}

Table refresh_tokens {
  id integer [primary key, increment]
  user_id integer [ref: > users.id, not null, note: 'User who owns this refresh token']
  token_hash varchar(255) [not null, unique, note: 'SHA256 hash of the refresh token']
  jwt_id varchar(255) [not null, note: 'JWT ID for token correlation']
  expires_at int [not null, note: 'Unix timestamp expiration (7 days)']
  revoked_at int [null, note: 'Unix timestamp when revoked']
  revoked_reason varchar(255) [null, note: 'Reason for revocation (audit)']
  replaced_by_token varchar(255) [null, note: 'Hash of replacement token (audit trail)']
  client_info varchar(255) [not null, note: 'Client identification']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  indexes {
    token_hash [unique, name: 'idx_refresh_tokens_token_hash']
    (user_id, revoked_at) [name: 'idx_refresh_tokens_user_revoked']
    expires_at [name: 'idx_refresh_tokens_expires_at']
    jwt_id [name: 'idx_refresh_tokens_jwt_id']
  }

  Note: 'Secure refresh token storage for JWT authentication with rotation'
}

Table two_factor_tokens {
  id integer [primary key, increment]
  user_id integer [ref: > users.id, not null, note: 'User who owns this 2FA token']
  token_hash varchar(255) [not null, unique, note: 'SHA256 hash of the verification code']
  token_type varchar(50) [not null, note: 'Type: email_verification, login_code']
  purpose varchar(100) [not null, note: 'Purpose: 2fa_setup, 2fa_login, account_verification']
  expires_at int [not null, note: 'Unix timestamp expiration (10 minutes for email codes)']
  used_at int [null, note: 'Unix timestamp when successfully used']
  attempts integer [default: 0, not null, note: 'Number of validation attempts made']
  max_attempts integer [default: 3, not null, note: 'Maximum allowed attempts']
  client_info varchar(255) [not null, note: 'Client identification for audit']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  indexes {
    user_id [name: 'idx_two_factor_tokens_user_id']
    token_hash [unique, name: 'idx_two_factor_tokens_hash']
    expires_at [name: 'idx_two_factor_tokens_expires_at']
    token_type [name: 'idx_two_factor_tokens_type']
  }

  Note: 'Time-limited verification tokens for 2FA email codes and setup processes'
}

Table groups {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  name varchar(200) [not null]
  description text
  created_by_user_id integer [ref: > users.id, not null]
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  Note: 'Groups for expense sharing - initially focused on couples (2 users)'
}

Table group_members {
  id integer [primary key, increment]
  group_id integer [ref: > groups.id, not null]
  user_id integer [ref: > users.id, not null]
  role_id int [not null, default: 'member', note: 'member = 2, admin = 1']
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  indexes {
    (group_id, user_id) [unique]
  }

  Note: 'Group membership - links users to expense groups'
}

Table expenses {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  group_id integer [ref: > groups.id, not null]
  title varchar(255) [not null]
  description text
  amount decimal(12,2) [not null, note: 'Total expense amount']
  paid_by_user_id integer [ref: > users.id, not null, note: 'User who paid the expense']
  expense_date date [not null, note: 'Date only (maps to .NET DateOnly)']
  category_id int [not null, note: 'Optional categorization']
  payment_mode_id int [not null, note: 'Payment method categorization']
  import_id integer [ref: > imports.id, null, note: 'Import that created this expense (null for manual)']
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  Note: 'Individual expenses within a group - can be imported or manually created'
}

Table expense_splits {
  id integer [primary key, increment]
  expense_id integer [ref: > expenses.id, not null]
  user_id integer [ref: > users.id, not null]
  split_amount decimal(12,2) [not null, note: 'Amount this user owes for this expense']
  created_at timestamp [not null, default: `now()`]
  updated_at timestamp [not null, default: `now()`]

  indexes {
    (expense_id, user_id) [unique]
  }

  Note: 'How each expense is split between users'
}

Table settlements {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  group_id integer [ref: > groups.id, not null]
  from_user_id integer [ref: > users.id, not null, note: 'User making the payment']
  to_user_id integer [ref: > users.id, not null, note: 'User receiving the payment']
  amount decimal(12,2) [not null, note: 'Settlement amount']
  settlement_date date [not null, note: 'Date only (maps to .NET DateOnly)']
  description text [note: 'Optional settlement description']
  deleted_at int [null, note: 'Soft delete unix timestamp']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  Note: 'Settlements between users to balance expenses'
}

Table imports {
  id integer [primary key, increment]
  guid uuid [unique, not null, note: 'Public identifier for frontend']
  user_id integer [ref: > users.id, not null, note: 'User who performed the import']
  group_id integer [ref: > groups.id, not null, note: 'Group where data was imported']
  filename varchar(255) [not null, note: 'Original backup filename']
  file_hash varchar(32) [not null, note: 'MD5 hash for duplicate detection']
  import_date date [not null, note: 'Date only (maps to .NET DateOnly)']
  records_count integer [not null, default: 0, note: 'Number of records processed']
  status_id integer [not null, default: 1, note: '1=Pending, 2=Processing, 3=Completed, 4=Failed']
  import_type_id integer [not null, note: '1=Cospend (enum ImportType)']
  error_details text [note: 'Error information if import failed']
  started_at int [null, note: 'Unix timestamp when processing started']
  completed_at int [null, note: 'Unix timestamp when processing completed']
  duration_ms int [null, note: 'Processing duration in milliseconds']
  created_at int [not null, note: 'Unix timestamp in seconds']
  updated_at int [not null, note: 'Unix timestamp in seconds']

  indexes {
    guid [unique, name: 'idx_imports_guid']
    (user_id, import_date) [name: 'idx_imports_user_date']
    (group_id, import_date) [name: 'idx_imports_group_date']
    (status_id, import_date) [name: 'idx_imports_status_date']
    (group_id, file_hash) [unique, name: 'idx_imports_group_hash']
  }

  Note: 'Track import operations with detailed status and type information'
}
